Esquema Relacional Supabase (PostgreSQL + PostGIS)

-- ESQUEMA BASE DE DATOS GO BIKE (SUPABASE)
CREATE TABLE usuarios (
  id UUID PRIMARY KEY REFERENCES auth.users(id),
  nombre TEXT NOT NULL,
  ciudad TEXT DEFAULT 'Bogotá D.C.',
  tipo_ciclismo TEXT CHECK (tipo_ciclismo IN ('Ruta','MTB','Urbano','Recreativo')),
  nivel TEXT CHECK (nivel IN ('Principiante','Intermedio','Avanzado')),
  descripcion TEXT,
  foto_url TEXT,
  km_totales NUMERIC DEFAULT 0,
  created_at TIMESTAMPTZ DEFAULT NOW()
);

CREATE TABLE rutas (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  creador_id UUID REFERENCES usuarios(id),
  nombre TEXT NOT NULL,
  zona TEXT NOT NULL,
  distancia_km NUMERIC NOT NULL,
  desnivel_m INT,
  dificultad TEXT CHECK (dificultad IN ('Baja','Media','Alta')),
  coordenadas JSONB NOT NULL,
  es_demo BOOLEAN DEFAULT FALSE
);

CREATE TABLE reportes_seguridad (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  usuario_id UUID REFERENCES usuarios(id),
  categoria TEXT NOT NULL,
  descripcion TEXT NOT NULL,
  ubicacion_texto TEXT,
  lat DOUBLE PRECISION NOT NULL,
  lng DOUBLE PRECISION NOT NULL,
  confirmaciones INT DEFAULT 1,
  expira_en TIMESTAMPTZ NOT NULL,
  estado TEXT DEFAULT 'activo',
  created_at TIMESTAMPTZ DEFAULT NOW()
);

CREATE TABLE publicaciones (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  usuario_id UUID REFERENCES usuarios(id),
  contenido TEXT NOT NULL,
  ruta_etiqueta TEXT,
  likes_count INT DEFAULT 0,
  created_at TIMESTAMPTZ DEFAULT NOW()
);

CREATE TABLE fotografias (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  publicacion_id UUID REFERENCES publicaciones(id) ON DELETE CASCADE,
  usuario_id UUID REFERENCES usuarios(id),
  storage_path TEXT NOT NULL
);

CREATE TABLE comentarios (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  publicacion_id UUID REFERENCES publicaciones(id) ON DELETE CASCADE,
  usuario_id UUID REFERENCES usuarios(id),
  texto TEXT NOT NULL,
  created_at TIMESTAMPTZ DEFAULT NOW()
);

CREATE TABLE seguidores (
  seguidor_id UUID REFERENCES usuarios(id),
  seguido_id UUID REFERENCES usuarios(id),
  PRIMARY KEY (seguidor_id, seguido_id)
);

CREATE TABLE rodadas (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  organizador_id UUID REFERENCES usuarios(id),
  titulo TEXT NOT NULL,
  fecha DATE NOT NULL,
  hora TIME NOT NULL,
  punto_encuentro TEXT NOT NULL,
  distancia_km NUMERIC NOT NULL,
  dificultad TEXT NOT NULL,
  cupos_totales INT NOT NULL
);

CREATE TABLE inscripciones (
  rodada_id UUID REFERENCES rodadas(id) ON DELETE CASCADE,
  usuario_id UUID REFERENCES usuarios(id),
  PRIMARY KEY (rodada_id, usuario_id)
);
